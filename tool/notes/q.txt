  const QrScanScreen({super.key});

  @override
  ConsumerState<QrScanScreen> createState() => _QrScanScreenState();
}

    _codeCtrl.dispose();
    super.dispose();
  }

    if (peerId == ownId) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('Das ist dein eigener Code!'),
            behavior: SnackBarBehavior.floating,
          ),
        );
      }
      return;
    }

        ),
